    (r ? r.chaves : []).forEach(function (k) { estado.viajados[k] = 1; });
    // primeiro nome: o da consulta (já vem pronto); se ela falhar, o digitado
    if (r && r.primeiro_nome) estado.primeiro = r.primeiro_nome;
    var nome = esc(estado.primeiro);
    var viajou = r ? r.viajou : [], vaiViajar = r ? r.vai_viajar : [];
    var da = r ? r.destino_anuncio : null;
    if (!da && destinoAnuncio) da = { chave: destinoAnuncio, nome: artigo(destinoAnuncio) };
    // Bruno 18/11 (UX): título curto + viagens em lista, em vez de uma frase comprida
    // Bruno 18/11: "Você se interessou pela Itália, que tal ela ser a próxima?"
    var alvo = '';
    if (da) {
      var m = /^(a|o|as|os)\s+(.+)$/.exec(da.nome);
      var ela = m ? { a: 'ela', o: 'ele', as: 'elas', os: 'eles' }[m[1]] : '';   // sem artigo (Portugal): sem pronome
      alvo = 'Você se interessou ' + porArtigo(da.nome) + '.<br>Que tal ' + (ela || esc(da.nome)) + ' ser a sua próxima viagem?';
    }
    var h, sub = 'Separamos estes roteiros para você:';
    if (viajou.length) {
      h = alvo || 'Qual vai ser a sua próxima viagem?';
      sub = 'Com base nas suas viagens, separamos estes roteiros:';
    } else if (alvo) h = alvo;
    else if (vaiViajar.length) h = 'Já pensando na próxima viagem?';
    else h = 'Que bom ter você de volta!';
    var temHist = !!(viajou.length || vaiViajar.length);
    $('[data-ola]').textContent = 'Olá, ' + estado.primeiro + '!';
    $('#pc-h3').innerHTML = h;
    var quando = function (d) { var mm = +String(d).slice(5, 7), ano = String(d).slice(0, 4); return mm >= 1 && mm <= 12 ? MESES[mm - 1] + ' de ' + ano : ''; };
    var item = function (v, ic) {
      return '<li><svg class="pc-ic" aria-hidden="true"><use href="#pc-i-' + ic + '"/></svg><span><b>' + esc(v.nome) + '</b>' + (quando(v.data) ? '<small>' + quando(v.data) + '</small>' : '') + '</span></li>';
    };
    var hist = $('[data-hist]');
    if (temHist) {
      $('[data-hist-t]').textContent = viajou.length ? 'Suas viagens com a Lielu' : 'Sua próxima viagem com a Lielu';
      $('[data-hist] .pc-historico').innerHTML = viajou.length ? viajou.map(function (v) { return item(v, 'check'); }).join('') : vaiViajar.map(function (v) { return item(v, 'aviao'); }).join('');
      hist.hidden = false;
    } else hist.hidden = true;
    var achada = r ? fotoDasViagens(r.cruas && r.cruas.length ? r.cruas : [r.recente]) : null, box = $('[data-lembra]');
    estado.nomeCompleto = (r && r.nome_completo) || '';
    if (achada) {
      var ft = achada.foto, mm = +ft.mes.slice(5, 7), ano = ft.mes.slice(0, 4);
      box.innerHTML = '<img src="' + esc(CONFIG.fotos_grupo + 'fotos/' + ft.foto) + '" alt="Foto do grupo da sua viagem" width="960" height="640" decoding="async">' +
        '<figcaption>Lembra dessa viagem?<small>' + esc(achada.viagem.nome || NOME_DESTINO[ft.chave] || '') + ' · ' + MESES[mm - 1] + ' de ' + ano + '</small></figcaption>';
      box.hidden = false;
      estado.fotoMostrada = ft.foto;
    } else { box.hidden = true; box.innerHTML = ''; estado.fotoMostrada = ''; }

    // Bruno 18/11: cada cartão diz POR QUE está ali; a lista tem um título claro
    var ref = da && NOME_DESTINO[da.chave] ? artigo(da.chave) : (viajou[0] ? (NOME_DESTINO[viajou[0].chave] ? artigo(viajou[0].chave) : viajou[0].nome) : '');
    estado.sugestoes = montaSugestoes(r);
    var temAnuncio = !!(destinoAnuncio && estado.sugestoes[0] && estado.sugestoes[0].destino === destinoAnuncio);
    // Bruno 18/11: volta o subtítulo da v8 (o título grande é o de cima); os selos dos cartões ficam
    $('[data-sub]').textContent = viajou.length ? 'Com base nas suas viagens, separamos estes roteiros:' : 'Separamos estes roteiros para você:';
    $('[data-lista="sugestoes"]').innerHTML = estado.sugestoes.map(function (s, i) {
      var selo = i === 0 && temAnuncio ? 'Você viu no anúncio' : (ref && s.destino !== (da && da.chave) ? 'Parecido com ' + ref : '');
      return cardHTML(s, i, selo, i === 0 && temAnuncio ? 'anuncio' : 'parecido');
    }).join('');
    $('[data-acao="outro"] .pc-letra').textContent = 'ABCD'.charAt(estado.sugestoes.length);
    log('tela 3', { historico: temHist, resposta: r, sugestoes: estado.sugestoes.map(function (s) { return s.slug; }) });
    vai('sugestoes');
  }

  function mensagem(s) {

  function mostraFinal(s, via) {

    if (ligar) ligar.hidden = !(CONFIG.modo_teste || (CONFIG.endpoint_ligacao && CONFIG.ligacao_no_ar));
    if (restaurando) { vai('final'); return; }
    salva({ escolha: s.slug, via: via });
    evento('pc_escolha', { roteiro: s.roteiro, saida: s.data_txt, via: via });
    postar(corpo('escolha', { roteiro: s.roteiro, saida: s.data_txt, saida_slug: s.slug, link_roteiro: s.link, destino_escolhido: s.destino, via: via, historico_mostrado: !!(estado.resposta && estado.resposta.achou && estado.resposta.nome_bateu) }));

  }
